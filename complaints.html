document.addEventListener('DOMContentLoaded', () => { const COLLECTION_ID = 'complaints'; // Éléments du DOM const tableBody = document.getElementById('complaints-table-body') || document.querySelector('tbody'); const searchInput = document.getElementById('search-input'); const filterStatus = document.getElementById('filter-status'); const btnOpenModal = document.getElementById('btn-open-modal'); // Injection dynamique de la modale de création injectComplaintModal(); const modal = document.getElementById('complaint-modal'); const modalClose = document.getElementById('modal-close'); const btnCancel = document.getElementById('btn-cancel-modal'); const formComplaint = document.getElementById('form-complaint'); let searchDebounceTimeout = null; // Chargement initial loadComplaints(); // --- Événements --- if (searchInput) { searchInput.addEventListener('input', () => { clearTimeout(searchDebounceTimeout); searchDebounceTimeout = setTimeout(loadComplaints, 300); }); } if (filterStatus) { filterStatus.addEventListener('change', loadComplaints); } if (btnOpenModal) btnOpenModal.addEventListener('click', openModal); if (modalClose) modalClose.addEventListener('click', closeModal); if (btnCancel) btnCancel.addEventListener('click', closeModal); if (formComplaint) { formComplaint.addEventListener('submit', async (e) => { e.preventDefault(); await handleSaveComplaint(); }); } // --- Fonctions --- /** * Charge et filtre la liste des réclamations depuis Appwrite. */ async function loadComplaints() { if (!tableBody) return; tableBody.innerHTML = 'Chargement des réclamations...'; try { const queries = [Appwrite.Query.orderDesc('$createdAt')]; const querySearch = searchInput ? searchInput.value.trim() : ''; const queryStatus = filterStatus ? filterStatus.value : ''; if (querySearch) { queries.push(Appwrite.Query.search('subject', querySearch)); } if (queryStatus) { queries.push(Appwrite.Query.equal('status', queryStatus)); } const response = await AppwriteConfig.databases.listDocuments( AppwriteConfig.databaseId, COLLECTION_ID, queries ); renderComplaints(response.documents); } catch (error) { console.error('Erreur lors du chargement des réclamations :', error); tableBody.innerHTML = `Erreur : ${Utils.escapeHtml(error.message)}`; } } /** * Rendu HTML des réclamations dans le tableau. */ function renderComplaints(complaints) { if (!complaints || complaints.length === 0) { tableBody.innerHTML = 'Aucune réclamation trouvée.'; return; } tableBody.innerHTML = complaints.map(item => { const isUrgent = item.priority === 'urgent'; return ` ${Utils.escapeHtml(item.subject || 'Sujet non spécifié')} ${Utils.escapeHtml(item.client_name || 'Client anonyme')} ${Utils.escapeHtml(item.priority || 'Moyenne')} ${Utils.escapeHtml(item.status || 'Nouveau')} ${Utils.formatDate(item.$createdAt)} `; }).join(''); } /** * Sauvegarde une nouvelle réclamation dans Appwrite. */ async function handleSaveComplaint() { const btnSubmit = formComplaint.querySelector('button[type="submit"]'); btnSubmit.disabled = true; const payload = { subject: document.getElementById('complaint-subject').value.trim(), client_name: document.getElementById('complaint-client').value.trim(), priority: document.getElementById('complaint-priority').value, status: document.getElementById('complaint-status').value, description: document.getElementById('complaint-desc').value.trim() }; try { await AppwriteConfig.databases.createDocument( AppwriteConfig.databaseId, COLLECTION_ID, Appwrite.ID.unique(), payload ); closeModal(); formComplaint.reset(); await loadComplaints(); } catch (error) { alert("Erreur lors de la création du ticket : " + error.message); } finally { btnSubmit.disabled = false; } } function openModal() { if (modal) modal.classList.add('active'); } function closeModal() { if (modal) modal.classList.remove('active'); } /** * Génère la modale de création de réclamation dans le DOM. */ function injectComplaintModal() { if (document.getElementById('complaint-modal')) return; const modalHTML = `

Nouveau Ticket de Réclamation

`; document.body.insertAdjacentHTML('beforeend', modalHTML); } });
