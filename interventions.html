document.addEventListener('DOMContentLoaded', () => { const COLLECTION_ID = 'interventions'; // Éléments du DOM const tableBody = document.getElementById('interventions-table-body') || document.querySelector('tbody'); const searchInput = document.getElementById('search-input'); const filterStatus = document.getElementById('filter-status'); const btnOpenModal = document.getElementById('btn-open-modal'); // Injection de la modale dans le DOM injectInterventionModal(); const modal = document.getElementById('intervention-modal'); const modalClose = document.getElementById('modal-close'); const btnCancel = document.getElementById('btn-cancel-modal'); const formIntervention = document.getElementById('form-intervention'); let searchDebounceTimeout = null; // Chargement initial loadInterventions(); // --- Événements --- if (searchInput) { searchInput.addEventListener('input', () => { clearTimeout(searchDebounceTimeout); searchDebounceTimeout = setTimeout(loadInterventions, 300); }); } if (filterStatus) { filterStatus.addEventListener('change', loadInterventions); } if (btnOpenModal) btnOpenModal.addEventListener('click', openModal); if (modalClose) modalClose.addEventListener('click', closeModal); if (btnCancel) btnCancel.addEventListener('click', closeModal); if (formIntervention) { formIntervention.addEventListener('submit', async (e) => { e.preventDefault(); await handleSaveIntervention(); }); } // --- Fonctions --- /** * Charge et filtre la liste des interventions depuis Appwrite. */ async function loadInterventions() { if (!tableBody) return; tableBody.innerHTML = 'Chargement des interventions...'; try { const queries = [Appwrite.Query.orderDesc('$createdAt')]; const querySearch = searchInput ? searchInput.value.trim() : ''; const queryStatus = filterStatus ? filterStatus.value : ''; if (querySearch) { queries.push(Appwrite.Query.search('title', querySearch)); } if (queryStatus) { queries.push(Appwrite.Query.equal('status', queryStatus)); } const response = await AppwriteConfig.databases.listDocuments( AppwriteConfig.databaseId, COLLECTION_ID, queries ); renderInterventions(response.documents); } catch (error) { console.error('Erreur de chargement des interventions :', error); tableBody.innerHTML = `Erreur : ${Utils.escapeHtml(error.message)}`; } } /** * Injecte les lignes du tableau. */ function renderInterventions(interventions) { if (!interventions || interventions.length === 0) { tableBody.innerHTML = 'Aucune intervention trouvée.'; return; } tableBody.innerHTML = interventions.map(item => { const isUrgent = item.priority === 'urgent'; return ` ${Utils.escapeHtml(item.title || 'Sans titre')} ${Utils.escapeHtml(item.client_name || 'Client non spécifié')} ${Utils.escapeHtml(item.priority || 'Normale')} ${Utils.escapeHtml(item.status || 'En attente')} ${Utils.formatDate(item.scheduled_at || item.$createdAt)} `; }).join(''); } /** * Sauvegarde une nouvelle intervention. */ async function handleSaveIntervention() { const btnSubmit = formIntervention.querySelector('button[type="submit"]'); btnSubmit.disabled = true; const payload = { title: document.getElementById('intervention-title').value.trim(), client_name: document.getElementById('intervention-client').value.trim(), priority: document.getElementById('intervention-priority').value, status: document.getElementById('intervention-status').value, scheduled_at: document.getElementById('intervention-date').value || new Date().toISOString(), description: document.getElementById('intervention-desc').value.trim() }; try { await AppwriteConfig.databases.createDocument( AppwriteConfig.databaseId, COLLECTION_ID, Appwrite.ID.unique(), payload ); closeModal(); formIntervention.reset(); await loadInterventions(); } catch (error) { alert("Erreur lors de la création de l'intervention : " + error.message); } finally { btnSubmit.disabled = false; } } function openModal() { if (modal) modal.classList.add('active'); } function closeModal() { if (modal) modal.classList.remove('active'); } /** * Structure HTML de la fenêtre modale. */ function injectInterventionModal() { if (document.getElementById('intervention-modal')) return; const modalHTML = `
