document.addEventListener('DOMContentLoaded', () => { const COLLECTION_ID = 'clients'; // Éléments du DOM const tableBody = document.getElementById('clients-table-body'); const searchInput = document.getElementById('search-input'); const btnOpenModal = document.getElementById('btn-open-modal'); // Injection dynamique de la fenêtre modale si absente du HTML injectClientModal(); const modal = document.getElementById('client-modal'); // Chargement initial des données loadClients(); // --- Gestionnaires d'événements --- // Recherche en temps réel avec Debounce (300ms) if (searchInput) { searchInput.addEventListener('input', (e) => { clearTimeout(searchDebounceTimeout); searchDebounceTimeout = setTimeout(() => { loadClients(e.target.value.trim()); }, 300); }); } // Ouverture / Fermeture de la modale if (btnOpenModal) btnOpenModal.addEventListener('click', openModal); if (modalClose) modalClose.addEventListener('click', closeModal); if (btnCancel) btnCancel.addEventListener('click', closeModal); // Soumission du formulaire if (formClient) { formClient.addEventListener('submit', async (e) => { e.preventDefault(); await handleSaveClient(); }); } // --- Fonctions Principales --- /** * Charge la liste des clients depuis Appwrite avec option de recherche. */ async function loadClients(searchQuery = '') { if (!tableBody) return; tableBody.innerHTML = 'Chargement des clients...'; try { const queries = []; if (searchQuery) { queries.push(Appwrite.Query.search('name', searchQuery)); } const response = await AppwriteConfig.databases.listDocuments( AppwriteConfig.databaseId, COLLECTION_ID, queries ); renderClients(response.documents); } catch (error) { console.error('Erreur lors du chargement des clients :', error); tableBody.innerHTML = `Erreur : ${Utils.escapeHtml(error.message)}`; } } /** * Rendu HTML du tableau des clients. */ function renderClients(clients) { if (!clients || clients.length === 0) { tableBody.innerHTML = 'Aucun client trouvé.'; return; } tableBody.innerHTML = clients.map(client => ` ${Utils.escapeHtml(client.name || 'N/A')} ${Utils.escapeHtml(client.email || '-')} ${Utils.escapeHtml(client.phone || '-')} ${Utils.escapeHtml(client.status || 'Actif')} `).join(''); } /** * Enregistrement d'un nouveau client dans la base de données Appwrite. */ async function handleSaveClient() { const btnSubmit = formClient.querySelector('button[type="submit"]'); btnSubmit.disabled = true; const payload = { name: document.getElementById('client-name').value.trim(), email: document.getElementById('client-email').value.trim(), phone: document.getElementById('client-phone').value.trim(), status: document.getElementById('client-status').value }; try { await AppwriteConfig.databases.createDocument( AppwriteConfig.databaseId, COLLECTION_ID, Appwrite.ID.unique(), payload ); closeModal(); formClient.reset(); await loadClients(); } catch (error) { alert("Erreur lors de la création du client : " + error.message); } finally { btnSubmit.disabled = false; } } function openModal() { if (modal) modal.classList.add('active'); } function closeModal() { if (modal) modal.classList.remove('active'); } /** * Injecte dynamiquement la fenêtre modale de création dans le DOM. */ function injectClientModal() { if (document.getElementById('client-modal')) return;
