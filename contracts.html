document.addEventListener('DOMContentLoaded', () => { const COLLECTION_ID = 'contracts'; // Éléments du DOM const tableBody = document.getElementById('contracts-table-body') || document.querySelector('tbody'); const searchInput = document.getElementById('search-input'); const filterStatus = document.getElementById('filter-status'); const btnOpenModal = document.getElementById('btn-open-modal'); // Injection dynamique de la modale de création injectContractModal(); const modal = document.getElementById('contract-modal'); const modalClose = document.getElementById('modal-close'); const btnCancel = document.getElementById('btn-cancel-modal'); const formContract = document.getElementById('form-contract'); let searchDebounceTimeout = null; // Chargement initial loadContracts(); // --- Événements --- if (searchInput) { searchInput.addEventListener('input', () => { clearTimeout(searchDebounceTimeout); searchDebounceTimeout = setTimeout(loadContracts, 300); }); } if (filterStatus) { filterStatus.addEventListener('change', loadContracts); } if (btnOpenModal) btnOpenModal.addEventListener('click', openModal); if (modalClose) modalClose.addEventListener('click', closeModal); if (btnCancel) btnCancel.addEventListener('click', closeModal); if (formContract) { formContract.addEventListener('submit', async (e) => { e.preventDefault(); await handleSaveContract(); }); } // --- Fonctions --- /** * Charge et filtre la liste des contrats depuis Appwrite. */ async function loadContracts() { if (!tableBody) return; tableBody.innerHTML = 'Chargement des contrats...'; try { const queries = [Appwrite.Query.orderDesc('$createdAt')]; const querySearch = searchInput ? searchInput.value.trim() : ''; const queryStatus = filterStatus ? filterStatus.value : ''; if (querySearch) { queries.push(Appwrite.Query.search('title', querySearch)); } if (queryStatus) { queries.push(Appwrite.Query.equal('status', queryStatus)); } const response = await AppwriteConfig.databases.listDocuments( AppwriteConfig.databaseId, COLLECTION_ID, queries ); renderContracts(response.documents); } catch (error) { console.error('Erreur lors du chargement des contrats :', error); tableBody.innerHTML = `Erreur : ${Utils.escapeHtml(error.message)}`; } } /** * Rendu HTML des contrats dans le tableau. */ function renderContracts(contracts) { if (!contracts || contracts.length === 0) { tableBody.innerHTML = 'Aucun contrat trouvé.'; return; } tableBody.innerHTML = contracts.map(item => { const formattedAmount = item.amount ? new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' }).format(item.amount) : '-'; return ` ${Utils.escapeHtml(item.title || 'Sans nom')} ${Utils.escapeHtml(item.client_name || 'Client non associé')} ${formattedAmount} ${Utils.escapeHtml(item.status || 'Actif')} ${Utils.formatDate(item.end_date || item.$createdAt)} `; }).join(''); } /** * Sauvegarde un nouveau contrat dans Appwrite. */ async function handleSaveContract() { const btnSubmit = formContract.querySelector('button[type="submit"]'); btnSubmit.disabled = true; const payload = { title: document.getElementById('contract-title').value.trim(), client_name: document.getElementById('contract-client').value.trim(), amount: parseFloat(document.getElementById('contract-amount').value) || 0, status: document.getElementById('contract-status').value, start_date: document.getElementById('contract-start').value || new Date().toISOString(), end_date: document.getElementById('contract-end').value || null }; try { await AppwriteConfig.databases.createDocument( AppwriteConfig.databaseId, COLLECTION_ID, Appwrite.ID.unique(), payload ); closeModal(); formContract.reset(); await loadContracts(); } catch (error) { alert("Erreur lors de la création du contrat : " + error.message); } finally { btnSubmit.disabled = false; } } function openModal() { if (modal) modal.classList.add('active'); } function closeModal() { if (modal) modal.classList.remove('active'); } /** * Génère la modale de création de contrat dans le DOM. */ function injectContractModal() { if (document.getElementById('contract-modal')) return; const modalHTML = `

Nouveau Contrat

`; document.body.insertAdjacentHTML('beforeend', modalHTML); } });
